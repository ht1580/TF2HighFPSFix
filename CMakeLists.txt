cmake_minimum_required(VERSION 3.20)
project(TF2HighFPSFix VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(TimingCoreTests tests/timing_tests.cpp)
target_include_directories(TimingCoreTests PRIVATE src)
add_test(NAME TimingCore COMMAND TimingCoreTests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Titanfall 2 requires a 64-bit build")
  endif()
  enable_language(RC)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
  add_library(TF2HighFPSFix SHARED src/plugin.cpp src/validation.cpp src/plugin.rc)
  target_include_directories(TF2HighFPSFix PRIVATE src)
  target_link_libraries(TF2HighFPSFix PRIVATE minhook bcrypt shell32)
  target_compile_definitions(TF2HighFPSFix PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(ModuleValidationTests tests/module_validation.cpp src/validation.cpp)
  target_include_directories(ModuleValidationTests PRIVATE src)
  target_compile_definitions(ModuleValidationTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ModuleValidationTests PRIVATE bcrypt minhook)
  set_target_properties(TF2HighFPSFix PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/release")
  if(MINGW)
    target_link_options(TF2HighFPSFix PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(TimingCoreTests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(ModuleValidationTests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(MSVC)
    set_property(TARGET TF2HighFPSFix minhook TimingCoreTests ModuleValidationTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(TF2HighFPSFix PRIVATE /W4 /permissive-)
  else()
    target_compile_options(TF2HighFPSFix PRIVATE -Wall -Wextra -Wpedantic -Wno-cast-function-type)
  endif()
  add_custom_command(TARGET TF2HighFPSFix POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/TF2HighFPSFix.ini" "$<TARGET_FILE_DIR:TF2HighFPSFix>/TF2HighFPSFix.ini")
endif()
